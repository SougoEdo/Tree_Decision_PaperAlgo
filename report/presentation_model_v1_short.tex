\section{What we do, and how we test it}
\begin{frame}{What we do, and how we test it}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Problem}
\begin{itemize}
\item weights of $n$ assets every $h$ days, from features $x_t$, with fees
\item a decision problem, not a forecast
\end{itemize}
\medskip
\textbf{Model}
\begin{itemize}
\item small tree $\to$ leaf scores $\to$ mean--variance optimizer with fees
\item loss: decision regret through the optimizer
\item splits: confirmed on the tree's own traded path, pruned on held-out weeks
\end{itemize}
\column{0.48\textwidth}
\textbf{Test}
\begin{itemize}
\item synthetic market, one feature $x$, planted threshold $d$
\item benchmark: \emph{ideal rule} = optimizer fed $\E[r\mid x]$
\item constant volatility; asset vs cash; weekly unless stated
\end{itemize}
\medskip
\textbf{Model v1}
\begin{itemize}
\item frozen 24 September 2026
\item 20{,}000+ fitted trees behind the numbers; validated on 100 datasets
\end{itemize}
\end{columns}
\medskip
\centering\small Equations: appendix.
\end{frame}

\section{How the tree learns, in plain words}

\begin{frame}{Every Monday: one number in, one decision out}
\begin{center}

  arr/.style={-{Stealth[length=2mm]}, thick}]
\node[box] (x) {\textbf{Monday}\\read one number $x$};
\node[box, right=of x] (q) {\textbf{the questions}\\``is $x$ below $\theta$?''};
\node[box, right=of q] (b) {\textbf{the leaf}\\its fixed reply};
\node[box, right=of b, fill=leaf!15] (w) {\textbf{the answer}\\asset\ \ or\ \ cash};
\node[box, right=of w] (h) {\textbf{hold it}\\for the week,\\then start again};

\begin{itemize}
\item The tree = the questions + one answer per leaf. Nothing else. Depth 3: at most 3 questions in a row, 8 leaves.
\item \textbf{Learning the tree} = choosing where to cut ($\theta$) and what each leaf answers.
\item In the test: $x$ = the planted feature, $d$ = the planted threshold.
\end{itemize}
\medskip
\begin{block}{Simplification used in this section}
No fees, no risk term: answer = all asset or all cash. Model v1 has both, but at $\lambda=1$ and 3 bp its positions are 0 or 1 anyway: the pictures stay faithful.
\end{block}
\end{frame}

\begin{itemize}

\begin{itemize}

\begin{frame}{Try every cut, keep the cheapest, then check it by trading it}
\begin{columns}[T]

\column{0.39\textwidth}
\small
\begin{itemize}

\begin{frame}{The training loop, and the three periods}
\begin{center}
\begin{tikzpicture}[x=0.6cm, every node/.style={font=\small}]

\fill[figorange!30] (11.25,0) rectangle (15,0.9);

\node at (13.1,0.45) {\textbf{checking} (4 years)};
\draw[|-|] (0,1.25) -- (15,1.25); \node[above] at (7.5,1.25) {the 15 training years: the tree is built here};
\node[above, font=\scriptsize] at (17.5,1.25) {frozen tree, never seen};
\node at (17.5,0.45) {\textbf{test} (5 years)};
\foreach \y in {0,5,10,15,20} {\draw (\y,0) -- (\y,-0.2); \node[below, font=\scriptsize] at (\y,-0.2) {year \y};}
\end{tikzpicture}
\end{center}
\footnotesize
\begin{enumerate}
\item \textbf{Start.} One leaf, no cut: asset or cash for every week, whichever costs less.
\item \textbf{Grow.} The cut that lowers the cost the most (previous frame); kept only if trading it over the fitting years raises the Sharpe ratio.
\item \textbf{Repeat.} Best cut inside every leaf; the one that lowers the total cost the most is tried, kept if the Sharpe check passes. Stop when no candidate passes, or at 3 levels of cuts (8 leaves). Leaf $\ge20$ weeks and 10\% of its parent.
\item \textbf{Check.} The last 4 training years, never used above. Every cut judged there, the tree traded with and without it: removed unless the cost falls and the Sharpe ratio rises; below the root, cost reduction $\ge2$ standard errors.
\item \textbf{Freeze.} Answer of each remaining leaf recomputed on all 15 training years. Nothing changes after this.
\item \textbf{Test.} The frozen tree traded on 5 further years it has never seen: the reported number.
\end{enumerate}
\end{frame}

\begin{frame}{The fee inside the fit is not the fee the strategy pays}
\begin{columns}[T]
\column{0.49\textwidth}
\begin{block}{During the fit of a leaf: frozen holdings}
\small
Optimizer run week by week from the holdings of the \emph{current} tree's path, fixed for the round. Its fee penalty prices the move from those holdings to the candidate decision. \textbf{Not} the fees the strategy being built would pay: that strategy reaches each week with its own holdings.
\end{block}
\column{0.49\textwidth}
\begin{block}{During the replay: holdings updated}
\small
The strategy is traded week after week: holdings updated from its own previous decision, drifted by the returns; its fees counted. \textbf{Real fees: known only after the replay.} The replay is where a strategy is measured: Sharpe check of a cut, pruning check, test.
\end{block}
\end{columns}
\medskip
\small
Side effect: a leaf score within 3 bp of the score that justifies the current holdings leaves the optimizer where it is (the freeze of E10). v2: no fee inside the fit; fees only where a strategy is really traded.
\end{frame}

\section{The planted-threshold experiment (E2--E11)}
\begin{frame}{The test-bed: a market with a planted answer (constant volatility)}

\begin{frame}{The decision: mean--variance with proportional fees}
\footnotesize
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Notation}
\begin{itemize}
\item $u_t$: holdings between time $\left(t-1\right)^+$ and $t^-$, $u_{t+1}=\dfrac{w_t\circ(1+r_t)}{1+r_t^\top w_t}$, $u_1=\1/n$
\item $c$: the tree's scores, in units of the return; cash $=0$ (only the score differences matter)
\item $\Sigma$: covariance of the $h$-day return $=h\times$ the covariance of the last 180 daily returns; $\sigma_h^2=h\hat\sigma^2$
\item $f$: fee per traded notional, 3 bp on the asset, 0 on cash; $\lambda=1$: risk aversion, weight of the $h$-day variance
\item $\W=\{w:\,0\le w_i\le\bar{w},\ \sum_iw_i=1\}$
\end{itemize}
\column{0.48\textwidth}
\textbf{Properties}
\begin{itemize}
\item strictly concave $\Rightarrow$ unique $w^\star$
\item $w^\star(c+\alpha\1)=w^\star(c)$: only score differences matter
\item exact solution: 5 states per weight ($0$, cap$\ =\bar{w}$, $u_i$, buy, sell) $\Rightarrow$ $5^n$ linear systems, best feasible one
\end{itemize}
\end{columns}
\[
w^\star_t(c;u_t,\Sigma)=\argmax_{w\in\W}\ \underbrace{c^\top w}_{\text{gain}}-\underbrace{f^\top|w-u_t|}_{\text{fees}}-\underbrace{\lambda\,w^\top\Sigma w}_{\text{risk}}
\]
\end{frame}

\begin{frame}{Asset against cash: the position window and the no-trade band}
\footnotesize
One risky asset (weight $w$, score $c$, weight held before the decision $u$) against cash (score 0, no fee, no variance); $\kappa=2\lambda\sigma_h^2$:
\[
w^\star=\argmax_{0\le w\le1}\ c\,w-f\,|w-u|-\tfrac{\kappa}{2}\,w^2
\]
\vspace{-2mm}
\begin{columns}[T]
\column{0.44\textwidth}
\textbf{Without fee} ($f=0$): $w^\star=c/\kappa$, clipped to $[0,\bar{w}]$.
\begin{itemize}
\item the position is linear in the score inside the \emph{position window} $0\le c\le\bar{w}\kappa$: all cash below 0, fully invested above $\kappa$
\item $\lambda=1$, weekly, $\sigma=1\%$ a day: $\kappa=2\cdot1\cdot5\cdot(1\%)^2=0.1\%$ per 5 days
\end{itemize}
\column{0.54\textwidth}
\textbf{With a fee} $f$ per traded notional: moving away from $u$ costs $f|w-u|$, so the score must beat the fee before anything moves:
\[
w^\star=\begin{cases}
\min\{(c-f)/\kappa,\,\bar{w}\} & c-\kappa u>f \ \text{(buy)}\\
\max\{(c+f)/\kappa,\,0\} & c-\kappa u<-f \ \text{(sell)}\\
u & |c-\kappa u|\le f \ \text{(no trade)}
\end{cases}
\]
\end{columns}
\textbf{The no-trade band} is $|c-\kappa u|\le f$: half-width $f=3$ bp around $\kappa u$, the score that would justify the position already held. A score inside it leaves the portfolio as it is, whatever it holds.
\begin{center}
\begin{tabular}{@{}lccc@{}}
\toprule
decision rate & window $\kappa$ & band half-width $f$ & band / window\\
\midrule
weekly ($h=5$) & 0.10\% & 0.03\% & 30\%\\
twice a week ($h=2$) & 0.04\% & 0.03\% & 75\%\\
\bottomrule
\end{tabular}
\end{center}
\textbf{v1}: leaf scores of $\pm0.3$--$0.5\%$ per 5 days sit far outside the window $\Rightarrow$ \alert{$w\in\{0,1\}$}; continuous sizing needs $\lambda\ge5$. But a score near 0 with a drifted $u$ sits inside the band, and the tree then holds that mix (E10).
\end{frame}

\begin{frame}{The loss: decision regret; the score of a leaf}
\begin{align*}
U_t(w)&=r_t^\top w-f^\top|w-u_t|-\lambda\,w^\top\Sigma_t w\\
\Reg_t(c)&=U_t\big(w^\star(r_t;u_t,\Sigma_t)\big)-U_t\big(w^\star(c;u_t,\Sigma_t)\big)\ \ge0
\end{align*}
\centerline{\small hindsight decision $-$ decision on $c$ (given the same holdings $u_t$)}
\[
\hat c_\ell=\argmin_{c\in[-b,b]^n}\ \sum_{t\in\ell}\Reg_t(c),\qquad b=10\%\ \text{per holding period, needed for bound + resolution}
\]
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Leaf score}
\begin{itemize}
\item one regret per date ($r_t$ depends on $t$), summed over the leaf's dates
\item bounded coordinate search: 3 passes $\times$ 7 values
\item not a forecast: same regret for any score on the right side
\end{itemize}
\column{0.48\textwidth}
\textbf{Consequences}
\begin{itemize}
\item fees $\Rightarrow$ regret depends on $u_t$ $\Rightarrow$ holdings are frozen for the round (no backpropagation) then replayed (drifted) 
\item only differences: relative returns, asset vs cash $\implies$ help to choose the right features
\item same $\Sigma_t$ as the optimizer: unseen risk not learnt
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Training: replay, two growth gates, out of sample pruning}
\begin{center}
\resizebox{0.9\textwidth}{!}{\begin{tikzpicture}[node distance=4mm and 5mm, every node/.style={font=\small},
  box/.style={draw, rounded corners=2pt, align=center, inner sep=4pt, fill=navy!5},
  arr/.style={-{Stealth[length=2mm]}, thick}]
\node[box] (rows) {rows $(x_t,\,r_t,\,\Sigma_t)$\\in time order};
\node[box, right=of rows] (grow) {grow on the first 75\%\\(steps 0--3)};
\node[box, right=of grow] (refit) {score search again\\on the final path (4)};
\node[box, right=of refit] (prune) {prune on the\\last 25\% (5)};
\node[box, right=of prune] (final) {score search\\on all rows (6)};
\node[box, right=of final] (use) {frozen tree: $x\mapsto\hat c(x)$\\$\to$ optimizer $\to$ weights};
\draw[arr] (rows) -- (grow); \draw[arr] (grow) -- (refit); \draw[arr] (refit) -- (prune); \draw[arr] (prune) -- (final); \draw[arr] (final) -- (use);
\end{tikzpicture}}
\end{center}
\footnotesize
\begin{enumerate}
\item[0.] root score from $u_t=\1/n$; \textbf{replay} (trade the scores date by date from $u_1$) $\Rightarrow$ $u_t$, Sharpe $S$; holdings frozen for the round
\item[1.] \textbf{regret gate}, per leaf: score search again from current $u_t$; candidates; children of the 3 best; keep iff $\Delta/N_\ell>10^{-6}$, $\Delta=L_\ell-L_{\mathrm{left}}-L_{\mathrm{right}}$ (regret per date, in $h$-day return)
\item[2.] \textbf{Sharpe gate}: rank all kept splits by $\Delta$; whole-tree replay of the first; accept iff $S'>S$, else next; one per round; none $\Rightarrow$ stop
\item[3.] replay the new tree $\Rightarrow$ new $u_t$; back to 1 (depth $\le3$; $\ge20$ rows and $\ge10\%$ of the parent per leaf)
\item[4.] score search again on every leaf, final path; kept iff $S$ does not fall
\item[5.] \textbf{pruning}, bottom-up, on the last 25\% (never used for growth): with vs without the split, both replayed;
$g_t=\Reg_t(\text{without})-\Reg_t(\text{with})$; keep iff $\bar g>10^{-6}$, $\ \bar g>k_p\,\mathrm{se}(\bar g)$ [$k_p=2$ below the root, $0$ at the root], checking Sharpe $\uparrow$
\item[6.] score search again on all training rows
\end{enumerate}
\textbf{Gates 1--2: in-sample. Gate 5: out-of-sample.}
\end{frame}

\section{Appendix B: the split-search options and the frozen configuration}
\begin{frame}{Split-search options: quantile candidates, refinement, soft splits}
\scriptsize
\textbf{Common}: admissible candidates $\theta_{(1)}<\dots<\theta_{(K)}$ (each side $\ge m_\ast=\max(m,\lceil\varphi N\rceil)$ rows; $m=20$, $\varphi=10\%$); cheap screen
\[
Q(\theta)=\sum_{t\in I}\Reg_t\big(c_{\mathrm{side}(t)}(\theta)\big),\qquad c_{\mathrm{left}}(\theta)=\mathrm{clip}\big(\mathrm{mean}\{r_t: x_t\le\theta\},\pm b\big),\ \textit{cf Theorem 1, } \text{idem right}
\]
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Deciles by rank + refinement} 
\begin{itemize}
\item keep ranks $\mathrm{round}(q(K-1))$, $q=0.1,\dots,0.9$: 9 screens instead of $K\approx N-2m_\ast$
\item refinement: screen every candidate between the two deciles around the best one, $\theta_\circ$; best of them $\theta_\bullet$
\item replace iff $Q(\theta_\circ)-Q(\theta_\bullet)>k\,\mathrm{se}$, $\ \mathrm{se}=\sqrt{|I|}\,\mathrm{std}_t(\delta_t)$, $k=1$
\item 3 best survivors ranked by $Q$; replay the first, then the second, then the third; keep iff Sharpe $\uparrow$
\end{itemize}
\column{0.48\textwidth}
\textbf{Soft split} (off in v1: $\kappa=0$)
\[
z_k=\frac{Q(\theta_k)-\min_lQ(\theta_l)}{\mathrm{se}_k},\qquad
\pi_k=\frac{e^{-z_k/\kappa}}{\sum_le^{-z_l/\kappa}}
\]
\begin{itemize}
\item all candidates kept, weighted; Gaussian variant $e^{-z^2/2\kappa^2}$
\item membership: left with $\mu(x)=\sum_{k:\,\theta_k\ge x}\pi_k$, right with $1-\mu(x)$; weights $\omega_{t\ell}$ multiply down the tree
\item leaf fit: $\min_c\sum_t\omega_{t\ell}\Reg_t(c)$; row score $\hat c(x_t)=\sum_\ell\omega_{t\ell}\hat c_\ell$: a ramp, not a jump
\item $\kappa\to0$: plain split
\end{itemize}
\end{columns}
\end{frame}
