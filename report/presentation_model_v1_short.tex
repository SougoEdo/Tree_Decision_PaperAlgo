\section{What we do, and how we test it}
\begin{frame}{What we do, and how we test it}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Problem}
\begin{itemize}
\item weights of $n$ assets every $h$ days, from features $x_t$, with fees
\item a decision problem, not a forecast
\end{itemize}
\medskip
\textbf{Model}
\begin{itemize}
\item small tree $\to$ leaf scores $\to$ mean--variance optimizer with fees
\item loss: decision regret through the optimizer
\item splits confirmed on the tree's own traded path
\item splits pruned on held-out weeks (regret and Sharpe)
\end{itemize}
\column{0.48\textwidth}
\textbf{Test}
\begin{itemize}
\item synthetic market, one feature $x$, planted threshold $d$
\item benchmark: \emph{ideal rule} = optimizer fed $\E[r\mid x]$
\item constant volatility; asset vs cash; weekly unless stated
\end{itemize}
\medskip
\textbf{Model v1}
\begin{itemize}
\item frozen 24 September 2026
\item over 20{,}000 fitted trees behind the numbers
\item validated on 100 datasets
\end{itemize}
\end{columns}
\medskip
\centering\small The equations are in the appendix.
\end{frame}

\section{How the tree learns, in plain words}

\begin{frame}{Every Monday: one number in, one decision out}
\begin{center}
\resizebox{\textwidth}{!}{%
\begin{tikzpicture}[node distance=5mm, every node/.style={font=\small},
  box/.style={draw, rounded corners=2pt, align=center, inner sep=5pt, fill=navy!6, minimum height=1.1cm},
  arr/.style={-{Stealth[length=2mm]}, thick}]
\node[box] (x) {\textbf{Monday}\\read one number $x$};
\node[box, right=of x] (q) {\textbf{the questions}\\``is $x$ below $\theta$?''};
\node[box, right=of q] (b) {\textbf{the leaf}\\its fixed reply};
\node[box, right=of b, fill=leaf!15] (w) {\textbf{the answer}\\asset\ \ or\ \ cash};
\node[box, right=of w] (h) {\textbf{hold it}\\for the week,\\then start again};
\draw[arr] (x) -- (q); \draw[arr] (q) -- (b); \draw[arr] (b) -- (w); \draw[arr] (w) -- (h);
\end{tikzpicture}}
\end{center}
\begin{itemize}
\item The tree \textbf{is} the list of questions and the answer in each leaf. Nothing else. Depth 3 means at most three questions in a row, so at most 8 leaves.
\item \textbf{Learning the tree} = choosing where to cut ($\theta$) and what each leaf answers.
\item In the test, $x$ is the planted feature with a planted threshold $d$.
\end{itemize}
\medskip
\begin{block}{Simplification used in this section}
No fees, no risk term: the answer is: all in the asset or all in cash. Model v1 has fees and a risk term, but at $\lambda=1$ and 3 bp its positions are 0 or 1 anyway, so the pictures below are faithful.
\end{block}
\end{frame}

\begin{frame}{What one cut does: sort the weeks by $x$ and look at their returns}
\begin{columns}[T]
\column{0.6\textwidth}
\centering
\includegraphics[width=0.98\textwidth]{deck_bigpicture_split}
\column{0.39\textwidth}
\footnotesize
\begin{itemize}
\item Every grey dot is one training week: $x$ on the Monday, the return of the week that followed. 567 weeks: the first 11 of the 15 training years (the last 4 are kept for the check).
\item The blue dots average the returns of the weeks in each interval of $x$ (width 0.25): the weeks lose on average left of the cut, gain right of it.
\item So leaf 1 answers \textbf{cash} and leaf 2 answers \textbf{asset}.
\item The whole model is one number, the cut, learnt from those weeks. This market was built with the true cut being at $d=0$.
\end{itemize}

{\scriptsize in brackets: the money lost by that answer}
\column{0.43\textwidth}
\footnotesize
\textbf{The rule for one leaf}
\begin{itemize}
\item its answer is \textbf{asset} if its weeks gain in total, \textbf{cash} otherwise;
\item its \textbf{cost} is the money lost on the weeks where that answer was wrong: losses taken in the asset, gains missed in cash.
\end{itemize}
\textbf{Read the table}
\begin{itemize}
\item no cut: the 8 weeks gain $+1.5\%$ together, so the single leaf answers asset every week and loses on weeks 1, 2, 4, 6: cost $5.0\%$;
\item cut at $x=0$: the left leaf loses $-3.5\%$ together, answers cash and only misses week 3; the right leaf gains $+5.0\%$, answers asset and only loses week 6: cost $1.5\%$;
\item the cut brings the cost from $5.0\%$ down to $1.5\%$: worth making.
\end{itemize}
{\scriptsize In the real model this cost is the \emph{decision regret}: hindsight decision minus the leaf's decision, week by week.}
\end{columns}
\end{frame}

\begin{frame}{Try every cut, keep the cheapest, then check it by trading it}
\begin{columns}[T]
\column{0.6\textwidth}
\centering
\includegraphics[width=0.98\textwidth]{deck_bigpicture_cost}
\column{0.39\textwidth}
\small
\begin{itemize}
\item About 450 cut positions are possible between the sorted weeks (each leaf must keep at least 57 weeks, 10\% of the 567). Each is scored by the cost of its two leaves; the cheapest wins: the red dot.
\item \textbf{Then the check by trading.} The rule is traded Monday after Monday over the fitting weeks, the first 11 of the 15 training years, starting from the money actually held. The cut is kept only if the \textbf{Sharpe ratio} of that trading is higher than without it.
\item v1 shortcut: score the 9 deciles first, then every cut between the two deciles around the best: about eight times fewer candidates, same result.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{The training loop, and the three periods}
\begin{center}
\begin{tikzpicture}[x=0.6cm, every node/.style={font=\small}]
\fill[navy!15] (0,0) rectangle (11.25,0.9);
\fill[figorange!30] (11.25,0) rectangle (15,0.9);
\fill[leaf!20] (15,0) rectangle (20,0.9);
\draw (0,0) rectangle (20,0.9);
\node at (5.6,0.45) {\textbf{fitting} (the first 11 years): choose the cuts};
\node at (13.1,0.45) {\textbf{checking} (4 years)};
\draw[|-|] (0,1.25) -- (15,1.25); \node[above] at (7.5,1.25) {the 15 training years};
\node at (17.5,0.45) {\textbf{test} (5 years)};
\foreach \y in {0,5,10,15,20} {\draw (\y,0) -- (\y,-0.2); \node[below, font=\scriptsize] at (\y,-0.2) {year \y};}
\end{tikzpicture}
\end{center}
\footnotesize
\begin{enumerate}
\item \textbf{Start} with one leaf and no cut: asset or cash for every week, whichever costs less.
\item \textbf{Grow.} Find the cut that lowers the cost the most (previous frame). Keep it only if trading the two leaves over the fitting years raises the Sharpe ratio.
\item \textbf{Repeat.} Each leaf is now a smaller set of weeks. Find the best cut inside every leaf; among these candidates take the one that lowers the total cost the most, and keep it if the Sharpe check passes. Stop when no candidate passes, or at 3 levels of cuts (8 leaves). A leaf keeps at least 20 weeks and 10\% of its parent leaf.
\item \textbf{Check.} The last 4 of the 15 training years were never used above. Every cut is now judged there, the tree being traded with and without it: the cut is removed if the cost does not fall or the Sharpe ratio does not rise; below the root the cost reduction must also be at least two standard errors.
\item \textbf{Freeze.} The answer of each remaining leaf is recomputed on all 15 training years.
\item \textbf{Test.} The frozen tree is traded on the 5 test years, which it has never seen. That is the number we report.
\end{enumerate}
\end{frame}

\begin{frame}{The fee inside the fit is not the fee the strategy pays}
\begin{columns}[T]
\column{0.49\textwidth}
\begin{block}{During the fit of a leaf: frozen holdings}
\small
The optimizer is run week by week from the holdings that the \emph{current} tree's path had that week, fixed for the whole round. Its fee penalty prices the move from those holdings to the candidate decision. It does \textbf{not} represent the actual fees that the strategy being built would pay for moving from one asset to the other (here between the asset and cash): that strategy would reach each week with its own holdings.
\end{block}
\column{0.49\textwidth}
\begin{block}{During the replay: holdings updated}
\small
The strategy is traded week after week: its holdings are updated from its own previous decision, drifted by the returns, and the fees it pays are counted. So the \textbf{real fees are known only after the replay}. The replay is where a strategy is measured: the Sharpe check of a cut, the pruning check, the test.
\end{block}
\end{columns}
\medskip
\small
Side effect of the frozen holdings: a leaf score within 3 bp of the one that justifies the current holdings leaves the optimizer where it is (the freeze of E10). Decided for v2: no fee inside the fit; fees only where a strategy is really traded.
\end{frame}

\begin{frame}{E10 in six points}
\small
\begin{itemize}
\item \textbf{The Sharpe ratio of the regime drift sets the precision.} A $\pm25\%$ regime is found within 0.25 of the threshold in 8 datasets out of 10, a $\pm5\%$ one in 1 out of 4, and then half the trees keep no split. Fees change almost nothing.
\item \textbf{Volatility acts through that Sharpe ratio}, until 32--48\% a year, where the risk term makes the ideal position gradual and the recovery drops below what the Sharpe ratio alone predicts.
\item \textbf{At $\lambda=5$ the split sits above the threshold, on purpose.} The ideal position is a ramp in $x$; the best single split cuts it in the middle, not at $d$.
\item \textbf{Second splits are gone.} At most 8\% of the datasets keep one.
\item \textbf{A fast feature needs a fast decision rate, and the fee bill is part of the price.} With a half-life of a few days, weekly decisions cannot follow the regime; the tree switches less often than the ideal rule, not more. Deciding twice a week recovers the threshold and most of the ideal Sharpe, at 21--36 moves a year: 0.6--1.1\% a year at 3 bp, about 0.06--0.10 of Sharpe, nearly the whole gap to the no-fee tree at half-lives of 2 to 4 days, less than half of it from 14 days on.
\item \textbf{The rest of the gap is the fee inside the training, which freezes one tree in five at twice a week.} A leaf score within 0.03\% of zero sits in the optimizer's no-trade band, and the tree then holds whatever it has, a drifted mix, for years: those trees earn 0.2 instead of 0.9. Without the fee in the training none freezes: the case for taking the fees out of the regret (v2, item 1).
\end{itemize}
\end{frame}

\begin{frame}{E11: the second market: a threshold for the drift and one for the volatility}
\footnotesize
The feature is the same daily Ornstein--Uhlenbeck process; the price now has two regimes for its drift and two for its volatility ($t$ in trading days):
\begin{gather*}
x_{t+1}=m+e^{-k}(x_t-m)+\eta_t,\qquad \eta_t\sim\mathcal N\big(0,\,s^2(1-e^{-2k})\big)\\
\log\frac{S_{t+1}}{S_t}=\mu_t-\tfrac12\sigma_t^2+\sigma_t\,\varepsilon_t,\qquad
\mu_t=\begin{cases}+\mu & x_t\ge d\\ -\mu & x_t<d\end{cases},\qquad
\sigma_t=\begin{cases}\rho\,\sigma & x_t\ge d_\sigma\\ \sigma & x_t<d_\sigma\end{cases}
\end{gather*}
$\sigma=1\%$ a day (16\% a year), $\rho=0.5$ (the calm regime at half the volatility), $\mu=e\sigma$ with $e=0.10$ (drift $\pm25\%$ a year), $d=0$; everything else as in the first market.
\begin{columns}[T]
\column{0.46\textwidth}
\textbf{Two placements of the volatility threshold}
\begin{itemize}
\item $d_\sigma=d$: the bad regime is the noisy one, the usual picture of markets
\item $d_\sigma=d+1$: the good regime is split into a stressed part, $0\le x<1$, and a calm part, $x\ge1$
\end{itemize}
\column{0.52\textwidth}
\textbf{What a correct tree should do}
\begin{itemize}
\item the expected return given $x$ does not depend on the volatility: the drift threshold stays at $d=0$
\item the volatility acts through the position, $w=(g_h(x)\mp f)/(2\lambda\sigma_h^2)$ clipped to $[0,1]$: at $\lambda=1$ all or nothing in both regimes, so the volatility threshold is invisible; at $\lambda=10$ about one half in the stressed good regime and one in the calm one, a second boundary at $x=1$ of size rather than of sign
\end{itemize}
\end{columns}
\medskip
Seven settings: constant volatility, $d_\sigma=d$, $d_\sigma=d+1$, each at $\lambda=1$ and $\lambda=10$, plus $d_\sigma=d+1$ at $\lambda=10$ with a 20-day covariance; decisions every 2, 5, 10 and 21 days, drifts $\pm13$ and $\pm25\%$ a year, 40 datasets per cell: 2{,}240 fits, on the base in E7 (23 September) and on model v1 in E11 (28 September).
\end{frame}

\begin{frame}{The decision: mean--variance with proportional fees}
\footnotesize
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Notation}
\begin{itemize}
\item $u_t$: holdings between time $\left(t-1\right)^+$ and $t^-$, $u_{t+1}=\dfrac{w_t\circ(1+r_t)}{1+r_t^\top w_t}$, $u_1=\1/n$
\item $c$: the tree's scores, in units of the return; cash $=0$ (only the score differences matter)
\item $\Sigma$: covariance of the $h$-day return $=h\times$ the covariance of the last 180 daily returns; $\sigma_h^2=h\hat\sigma^2$
\item $f$: fee per traded notional, 3 bp on the asset, 0 on cash; $\lambda=1$: risk aversion, weight of the $h$-day variance
\item $\W=\{w:\,0\le w_i\le\bar{w},\ \sum_iw_i=1\}$
\end{itemize}
\column{0.48\textwidth}
\textbf{Properties}
\begin{itemize}
\item strictly concave $\Rightarrow$ unique $w^\star$
\item $w^\star(c+\alpha\1)=w^\star(c)$: only score differences matter
\item exact solution: 5 states per weight ($0$, cap$\ =\bar{w}$, $u_i$, buy, sell) $\Rightarrow$ $5^n$ linear systems, best feasible one
\end{itemize}
\end{columns}
\[
w^\star_t(c;u_t,\Sigma)=\argmax_{w\in\W}\ \underbrace{c^\top w}_{\text{gain}}-\underbrace{f^\top|w-u_t|}_{\text{fees}}-\underbrace{\lambda\,w^\top\Sigma w}_{\text{risk}}
\]
\end{frame}

\begin{frame}{Asset against cash: the position window and the no-trade band}
\footnotesize
One risky asset (weight $w$, score $c$, weight held before the decision $u$) against cash (score 0, no fee, no variance); $\kappa=2\lambda\sigma_h^2$:
\[
w^\star=\argmax_{0\le w\le1}\ c\,w-f\,|w-u|-\tfrac{\kappa}{2}\,w^2
\]
\vspace{-2mm}
\begin{columns}[T]
\column{0.44\textwidth}
\textbf{Without fee} ($f=0$): $w^\star=c/\kappa$, clipped to $[0,\bar{w}]$.
\begin{itemize}
\item the position is linear in the score inside the \emph{position window} $0\le c\le\bar{w}\kappa$: all cash below 0, fully invested above $\kappa$
\item $\lambda=1$, weekly, $\sigma=1\%$ a day: $\kappa=2\cdot1\cdot5\cdot(1\%)^2=0.1\%$ per 5 days
\end{itemize}
\column{0.54\textwidth}
\textbf{With a fee} $f$ per traded notional: moving away from $u$ costs $f|w-u|$, so the score must beat the fee before anything moves:
\[
w^\star=\begin{cases}
\min\{(c-f)/\kappa,\,\bar{w}\} & c-\kappa u>f \ \text{(buy)}\\
\max\{(c+f)/\kappa,\,0\} & c-\kappa u<-f \ \text{(sell)}\\
u & |c-\kappa u|\le f \ \text{(no trade)}
\end{cases}
\]
\end{columns}
\textbf{The no-trade band} is $|c-\kappa u|\le f$: half-width $f=3$ bp around $\kappa u$, the score that would justify the position already held. A score inside it leaves the portfolio as it is, whatever it holds.
\begin{center}
\begin{tabular}{@{}lccc@{}}
\toprule
decision rate & window $\kappa$ & band half-width $f$ & band / window\\
\midrule
weekly ($h=5$) & 0.10\% & 0.03\% & 30\%\\
twice a week ($h=2$) & 0.04\% & 0.03\% & 75\%\\
\bottomrule
\end{tabular}
\end{center}
\textbf{v1}: leaf scores of $\pm0.3$--$0.5\%$ per 5 days sit far outside the window $\Rightarrow$ \alert{$w\in\{0,1\}$}; continuous sizing needs $\lambda\ge5$. But a score near 0 with a drifted $u$ sits inside the band, and the tree then holds that mix (E10).
\end{frame}

\begin{frame}{The loss: decision regret; the score of a leaf}
\begin{align*}
U_t(w)&=r_t^\top w-f^\top|w-u_t|-\lambda\,w^\top\Sigma_t w\\
\Reg_t(c)&=U_t\big(w^\star(r_t;u_t,\Sigma_t)\big)-U_t\big(w^\star(c;u_t,\Sigma_t)\big)\ \ge0
\end{align*}
\centerline{\small hindsight decision $-$ decision on $c$ (given the same holdings $u_t$)}
\[
\hat c_\ell=\argmin_{c\in[-b,b]^n}\ \sum_{t\in\ell}\Reg_t(c),\qquad b=10\%\ \text{per holding period, needed for bound + resolution}
\]
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Leaf score}
\begin{itemize}
\item one regret per date ($r_t$ depends on $t$), summed over the leaf's dates
\item bounded coordinate search: 3 passes $\times$ 7 values
\item not a forecast: same regret for any score on the right side
\end{itemize}
\column{0.48\textwidth}
\textbf{Consequences}
\begin{itemize}
\item fees $\Rightarrow$ regret depends on $u_t$ $\Rightarrow$ holdings are frozen for the round (no backpropagation) then replayed (drifted) 
\item only differences: relative returns, asset vs cash $\implies$ help to choose the right features
\item same $\Sigma_t$ as the optimizer: unseen risk not learnt
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Training: replay, two growth gates, out of sample pruning}
\begin{center}
\resizebox{0.9\textwidth}{!}{\begin{tikzpicture}[node distance=4mm and 5mm, every node/.style={font=\small},
  box/.style={draw, rounded corners=2pt, align=center, inner sep=4pt, fill=navy!5},
  arr/.style={-{Stealth[length=2mm]}, thick}]
\node[box] (rows) {rows $(x_t,\,r_t,\,\Sigma_t)$\\in time order};
\node[box, right=of rows] (grow) {grow on the first 75\%\\(steps 0--3)};
\node[box, right=of grow] (refit) {score search again\\on the final path (4)};
\node[box, right=of refit] (prune) {prune on the\\last 25\% (5)};
\node[box, right=of prune] (final) {score search\\on all rows (6)};
\node[box, right=of final] (use) {frozen tree: $x\mapsto\hat c(x)$\\$\to$ optimizer $\to$ weights};
\draw[arr] (rows) -- (grow); \draw[arr] (grow) -- (refit); \draw[arr] (refit) -- (prune); \draw[arr] (prune) -- (final); \draw[arr] (final) -- (use);
\end{tikzpicture}}
\end{center}
\footnotesize
\begin{enumerate}
\item[0.] root score from $u_t=\1/n$; \textbf{replay} (trade the scores date by date from $u_1$) $\Rightarrow$ $u_t$, Sharpe $S$; holdings frozen for the round
\item[1.] \textbf{regret gate}, per leaf: score search again from current $u_t$; candidates; children of the 3 best; keep iff $\Delta/N_\ell>10^{-6}$, $\Delta=L_\ell-L_{\mathrm{left}}-L_{\mathrm{right}}$ (regret per date, in $h$-day return)
\item[2.] \textbf{Sharpe gate}: rank all kept splits by $\Delta$; whole-tree replay of the first; accept iff $S'>S$, else next; one per round; none $\Rightarrow$ stop
\item[3.] replay the new tree $\Rightarrow$ new $u_t$; back to 1 (depth $\le3$; $\ge20$ rows and $\ge10\%$ of the parent per leaf)
\item[4.] score search again on every leaf, final path; kept iff $S$ does not fall
\item[5.] \textbf{pruning}, bottom-up, on the last 25\% (never used for growth): with vs without the split, both replayed;
$g_t=\Reg_t(\text{without})-\Reg_t(\text{with})$; keep iff $\bar g>10^{-6}$, $\ \bar g>k_p\,\mathrm{se}(\bar g)$ [$k_p=2$ below the root, $0$ at the root], checking Sharpe $\uparrow$
\item[6.] score search again on all training rows
\end{enumerate}
\textbf{Gates 1--2: in-sample. Gate 5: out-of-sample.}
\end{frame}

\section{Appendix B: the split-search options and the frozen configuration}
\begin{frame}{Split-search options: quantile candidates, refinement, soft splits}
\scriptsize
\textbf{Common}: admissible candidates $\theta_{(1)}<\dots<\theta_{(K)}$ (each side $\ge m_\ast=\max(m,\lceil\varphi N\rceil)$ rows; $m=20$, $\varphi=10\%$); cheap screen
\[
Q(\theta)=\sum_{t\in I}\Reg_t\big(c_{\mathrm{side}(t)}(\theta)\big),\qquad c_{\mathrm{left}}(\theta)=\mathrm{clip}\big(\mathrm{mean}\{r_t: x_t\le\theta\},\pm b\big),\ \textit{cf Theorem 1, } \text{idem right}
\]
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Deciles by rank + refinement} 
\begin{itemize}
\item keep ranks $\mathrm{round}(q(K-1))$, $q=0.1,\dots,0.9$: 9 screens instead of $K\approx N-2m_\ast$
\item refinement: screen every candidate between the two deciles around the best one, $\theta_\circ$; best of them $\theta_\bullet$
\item replace iff $Q(\theta_\circ)-Q(\theta_\bullet)>k\,\mathrm{se}$, $\ \mathrm{se}=\sqrt{|I|}\,\mathrm{std}_t(\delta_t)$, $k=1$
\item 3 best survivors ranked by $Q$; replay the first, then the second, then the third; keep iff Sharpe $\uparrow$
\end{itemize}
\column{0.48\textwidth}
\textbf{Soft split} (off in v1: $\kappa=0$)
\[
z_k=\frac{Q(\theta_k)-\min_lQ(\theta_l)}{\mathrm{se}_k},\qquad
\pi_k=\frac{e^{-z_k/\kappa}}{\sum_le^{-z_l/\kappa}}
\]
\begin{itemize}
\item all candidates kept, weighted; Gaussian variant $e^{-z^2/2\kappa^2}$
\item membership: left with $\mu(x)=\sum_{k:\,\theta_k\ge x}\pi_k$, right with $1-\mu(x)$; weights $\omega_{t\ell}$ multiply down the tree
\item leaf fit: $\min_c\sum_t\omega_{t\ell}\Reg_t(c)$; row score $\hat c(x_t)=\sum_\ell\omega_{t\ell}\hat c_\ell$: a ramp, not a jump
\item $\kappa\to0$: plain split
\end{itemize}
\end{columns}
\end{frame}
